# Anime Role Detect - 技能化架构设计方案

---

## 1. 项目背景与目标

### 1.1 当前项目分析

**项目名称**: Anime Role Detect (角色分类系统)

**核心功能**:
- 基于深度学习的动漫/游戏角色识别
- 支持多模型（MobileNetV2, EfficientNet-B0/B3, ResNet50）
- 端到端工作流（数据采集→清洗→训练→推理）
- RESTful API + WebSocket 实时识别
- Docker/K8s 部署

**现有脚本分类**:
| 类别 | 脚本数量 | 功能说明 |
|------|---------|----------|
| `data_collection` | 40+ | 数据采集、爬虫、URL 管理 |
| `data_cleaning` | 30+ | 数据清洗、去重、质量检查 |
| `analysis` | 15+ | 数据分析、特征质量、URL 分布 |
| `model_training` | 10+ | 模型训练、管理 |
| `model_testing` | 20+ | 模型测试、性能基准 |
| `classification` | 5+ | 角色分类推理 |
| `api` | 3+ | API 服务（飞书、WebSocket） |
| `search` | 5+ | 以图搜图服务 |

### 1.2 技能化改造目标

借鉴 SkillHub 的设计理念，将现有脚本系统改造为：

1. **模块化技能包**：将脚本按功能打包成独立技能
2. **统一配置管理**：类似 `.env` 的多环境配置
3. **API 标准化**：统一的 API 调用接口
4. **技能编排**：支持技能组合和流程编排
5. **版本管理**：技能版本控制和更新机制

### 3.1 ardc-client - 核心客户端

#### 3.1.1 环境配置 (`load_env`)

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
统一环境配置加载
"""
import os
from pathlib import Path

def load_env(env_path=None):
    """
    加载 .env 文件，支持多环境
    
    查找优先级：
    1. 显式 env_path 参数
    2. ARDC_PROJECT_ENV_FILE 环境变量
    3. <cwd>/.env.<ARDC_ENV> → <cwd>/config/.env.<ARDC_ENV>
    4. <cwd>/.env → <cwd>/config/.env
    """
    env_name = os.environ.get('ARDC_ENV', '')
    project_env_file = os.environ.get('ARDC_PROJECT_ENV_FILE', '')
    
    env = {
        'ARDC_BASE_URL': os.environ.get('ARDC_BASE_URL', ''),
        'ARDC_API_KEY': os.environ.get('ARDC_API_KEY', ''),
        'ARDC_TOKEN': os.environ.get('ARDC_TOKEN', ''),
    }
    
    # 加载文件配置
    env_paths = []
    if env_path:
        env_paths.append(env_path)
    elif project_env_file:
        env_paths.append(project_env_file)
    else:
        if env_name:
            env_paths.append(f'.env.{env_name}')
            env_paths.append(f'config/.env.{env_name}')
        env_paths.append('.env')
        env_paths.append('config/.env')
    
    for candidate in env_paths:
        if os.path.exists(candidate):
            with open(candidate) as f:
                for line in f:
                    line = line.strip()
                    if '=' in line and not line.startswith('#'):
                        k, v = line.split('=', 1)
                        env[k.strip()] = v.strip().strip('"').strip("'")
            break
    
    return env
```

#### 3.1.2 API 调用封装

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
统一 API 调用接口
"""
import requests
import json

def call_api(base_url, token, endpoint, body=None):
    """
    调用 Anime Role Detect API
    
    Args:
        base_url: 服务地址 (如 http://localhost:8080)
        token: API Token
        endpoint: API 端点 (如 /api/classify)
        body: 请求体
    
    Returns:
        JSON 响应
    """
    headers = {
        'Authorization': f'Bearer {token}',
        'Content-Type': 'application/json',
    }
    
    resp = requests.post(
        f"{base_url}{endpoint}",
        json=body or {},
        headers=headers,
        timeout=(10, 60),
    )
    
    return resp.json()

def classify_image(base_url, token, image_path, model='efficientnet_b3'):
    """
    调用分类 API
    
    Args:
        image_path: 图片路径
        model: 模型名称
    
    Returns:
        分类结果
    """
    with open(image_path, 'rb') as f:
        files = {'image': f}
        data = {'model': model}
        
        resp = requests.post(
            f"{base_url}/api/classify",
            files=files,
            data=data,
            headers={'Authorization': f'Bearer {token}'},
        )
        
        return resp.json()
```

#### 3.2.2 采集脚本实现

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
批量采集角色图片
"""
import sys
import os
sys.path.insert(0, os.path.join(os.path.dirname(__file__), '..', '..', 'ardc-client', 'scripts'))

from ardc_client import load_env, call_api
from common.download_utils import download_image
from common.logging_config import setup_logger

logger = setup_logger(__name__)

def collect_images(role, count=100, output_dir='./data'):
    """
    批量采集角色图片
    
    Args:
        role: 角色名
        count: 目标数量
        output_dir: 输出目录
    """
    env = load_env()
    base_url = env.get('ARDC_BASE_URL', 'http://localhost:8080')
    token = env.get('ARDC_TOKEN', '')
    
    # 调用采集 API
    result = call_api(base_url, token, '/api/collect', {
        'role': role,
        'count': count,
        'output_dir': output_dir,
    })
    
    logger.info(f"采集完成：{result}")
    return result

if __name__ == '__main__':
    import argparse
    parser = argparse.ArgumentParser()
    parser.add_argument('--role', required=True, help='角色名')
    parser.add_argument('--count', type=int, default=100, help='目标数量')
    parser.add_argument('--output', default='./data', help='输出目录')
    
    args = parser.parse_args()
    collect_images(args.role, args.count, args.output)
```

#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
模型训练技能
"""
import sys
import os
sys.path.insert(0, os.path.join(os.path.dirname(__file__), '..', '..', 'ardc-client', 'scripts'))

from model_training.CharacterClassifier import CharacterClassifier
from common.logging_config import setup_logger

logger = setup_logger(__name__)

def train_model(
    data_dir,
    model_name='efficientnet_b3',
    epochs=50,
    batch_size=32,
    learning_rate=0.001,
    output_dir='./models'
):
    """
    训练角色分类模型
    
    Args:
        data_dir: 数据集目录
        model_name: 模型名称
        epochs: 训练轮数
        batch_size: 批次大小
        learning_rate: 学习率
        output_dir: 模型输出目录
    """
    classifier = CharacterClassifier(
        model_name=model_name,
        num_classes=None,  # 自动从数据集推断
    )
    
    classifier.train(
        train_dir=os.path.join(data_dir, 'train'),
        val_dir=os.path.join(data_dir, 'val'),
        epochs=epochs,
        batch_size=batch_size,
        learning_rate=learning_rate,
        save_path=os.path.join(output_dir, f'{model_name}.pth'),
    )
    
    logger.info(f"训练完成：{output_dir}/{model_name}.pth")

if __name__ == '__main__':
    import argparse
    parser = argparse.ArgumentParser()
    parser.add_argument('--data-dir', required=True, help='数据集目录')
    parser.add_argument('--model', default='efficientnet_b3', help='模型名称')
    parser.add_argument('--epochs', type=int, default=50, help='训练轮数')
    parser.add_argument('--batch-size', type=int, default=32, help='批次大小')
    parser.add_argument('--lr', type=float, default=0.001, help='学习率')
    parser.add_argument('--output', default='./models', help='输出目录')
    
    args = parser.parse_args()
    train_model(
        args.data_dir,
        args.model,
        args.epochs,
        args.batch_size,
        args.lr,
        args.output,
    )

### 4.2 技能编排工具

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
技能编排工具 - 组合多个技能执行复杂任务
"""
import yaml
import subprocess
from pathlib import Path

def run_workflow(workflow_file):
    """
    执行工作流
    
    Args:
        workflow_file: YAML 工作流文件
    """
    with open(workflow_file) as f:
        workflow = yaml.safe_load(f)
    
    for step in workflow['steps']:
        skill = step['skill']
        script = step['script']
        args = step.get('args', {})
        
        # 构建命令
        cmd = [
            'python',
            Path.home() / '.ardc' / 'skills' / skill / 'scripts' / script,
        ]
        
        for k, v in args.items():
            cmd.extend([f'--{k}', str(v)])
        
        # 执行
        print(f"执行步骤：{step.get('name', script)}")
        subprocess.run(cmd)

if __name__ == '__main__':
    import argparse
    parser = argparse.ArgumentParser()
    parser.add_argument('--workflow', required=True, help='工作流文件')
    args = parser.parse_args()
    
    run_workflow(args.workflow)
```
